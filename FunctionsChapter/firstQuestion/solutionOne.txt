import random

def is_valid_input(user_input):
    return user_input.isdigit() and 1 <= int(user_input) <= 100

def get_user_guess():
    while True:
        guess = input("Enter your guess (1-100): ")
        if is_valid_input(guess):
            return int(guess)
        print("Oops! That doesn't seem right. Please enter a number between 1 and 100.")

def start_guessing_game():
    secret_number = random.randint(1, 100)
    attempts = 0

    print("\n Welcome to the Number Guessing Game!")
    print("I've picked a number between 1 and 100. Try to guess it!\n")

    while True:
        guess = get_user_guess()
        attempts += 1

        if guess < secret_number:
            print("Too low! Try again.")
        elif guess > secret_number:
            print("Too high! Give it another shot.")
        else:
            printf("Congrats! You guessed it in {attempts} attempts. Well done!")
            break

if __name__ == "__main__":
    start_guessing_game()
